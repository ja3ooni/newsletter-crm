Getting Started

Choose your preferred API interface from the tabs above. The REST API provides traditional HTTP endpoints with comprehensive OpenAPI documentation, while the GraphQL API offers a more flexible query interface with real-time capabilities. 

# Base URLs
Production:  https://api.datatechtoncrm.com/v1
Staging:     https://staging-api.datatechtoncrm.com/v1
Development: http://localhost:8000/v1

# Authentication
Authorization: Bearer <jwt_token>
X-API-Key: <api_key>

API Usage Examples

Authentication

# Login to get JWT token
curl -X POST https://api.datatechtoncrm.com/v1/users/login \
  -H "Content-Type: application/json" \
  -d '{
    "email": "user@example.com",
    "password": "your-password"
  }'

# Response
{
  "accessToken": "eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9...",
  "refreshToken": "eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9...",
  "expiresIn": 3600,
  "user": {
    "id": "123e4567-e89b-12d3-a456-426614174000",
    "email": "user@example.com",
    "firstName": "John",
    "lastName": "Doe"
  }
}

Create a Newsletter

# REST API
curl -X POST https://api.datatechtoncrm.com/v1/newsletters \
  -H "Authorization: Bearer YOUR_JWT_TOKEN" \
  -H "Content-Type: application/json" \
  -d '{
    "title": "Weekly Tech Update",
    "templateId": "template-uuid",
    "content": {
      "sections": [
        {
          "id": "section-1",
          "type": "news",
          "title": "Latest News",
          "items": [
            {
              "id": "item-1",
              "title": "AI Breakthrough in Healthcare",
              "summary": "New AI model shows promising results...",
              "url": "https://example.com/article",
              "source": "TechNews"
            }
          ],
          "order": 1
        }
      ]
    },
    "segments": ["segment-uuid-1", "segment-uuid-2"]
  }'

GraphQL Query Example

# Get newsletters with analytics
query GetNewsletters($first: Int, $filters: NewsletterFiltersInput) {
  newsletters(pagination: { first: $first }, filters: $filters) {
    edges {
      node {
        id
        title
        status
        createdAt
        metrics {
          sent
          delivered
          openRate
          clickRate
        }
        analytics(timeRange: "7d") {
          topLinks {
            url
            clicks
            clickRate
          }
          deviceBreakdown {
            desktop
            mobile
            tablet
          }
        }
      }
    }
    pageInfo {
      hasNextPage
      endCursor
    }
    totalCount
  }
}

Create and Manage Contacts

# Create a new contact
curl -X POST https://api.datatechtoncrm.com/v1/crm/contacts \
  -H "Authorization: Bearer YOUR_JWT_TOKEN" \
  -H "Content-Type: application/json" \
  -d '{
    "email": "contact@example.com",
    "firstName": "Jane",
    "lastName": "Smith",
    "company": "Acme Corp",
    "jobTitle": "Marketing Manager",
    "tags": ["lead", "enterprise"],
    "customFields": {
      "industry": "Technology",
      "company_size": "500-1000"
    },
    "preferences": {
      "emailFrequency": "weekly",
      "contentTypes": ["news", "research"],
      "timezone": "America/New_York"
    }
  }'

# Update contact lead score
curl -X PUT https://api.datatechtoncrm.com/v1/crm/contacts/contact-uuid \
  -H "Authorization: Bearer YOUR_JWT_TOKEN" \
  -H "Content-Type: application/json" \
  -d '{
    "customFields": {
      "lead_score": 85,
      "last_interaction": "2024-01-15T10:30:00Z"
    }
  }'

Create Dynamic Segments

# Create a segment for engaged users
curl -X POST https://api.datatechtoncrm.com/v1/crm/segments \
  -H "Authorization: Bearer YOUR_JWT_TOKEN" \
  -H "Content-Type: application/json" \
  -d '{
    "name": "Highly Engaged Users",
    "description": "Users with high engagement scores",
    "conditions": [
      {
        "field": "leadScore",
        "operator": "greater_than",
        "value": 70
      },
      {
        "field": "lifecycle",
        "operator": "in",
        "value": ["lead", "customer"],
        "logicalOperator": "AND"
      },
      {
        "field": "tags",
        "operator": "contains",
        "value": "engaged",
        "logicalOperator": "AND"
      }
    ],
    "isAutoUpdating": true
  }'

Send Newsletter with Analytics

# Send newsletter to specific segments
curl -X POST https://api.datatechtoncrm.com/v1/newsletters/newsletter-uuid/send \
  -H "Authorization: Bearer YOUR_JWT_TOKEN" \
  -H "Content-Type: application/json" \
  -d '{
    "segments": ["segment-uuid-1", "segment-uuid-2"],
    "scheduledAt": "2024-01-20T09:00:00Z"
  }'

# Response
{
  "jobId": "job_abc123",
  "status": "queued",
  "estimatedDelivery": "2024-01-20T09:05:00Z",
  "recipientCount": 1250
}

# Check sending progress
curl -X GET https://api.datatechtoncrm.com/v1/newsletters/newsletter-uuid/analytics \
  -H "Authorization: Bearer YOUR_JWT_TOKEN"

Real-time Updates with GraphQL Subscriptions

# Subscribe to newsletter sending progress
subscription NewsletterProgress($newsletterId: UUID!) {
  newsletterSendProgress(newsletterId: $newsletterId) {
    newsletterId
    status
    progress
    sent
    failed
    estimatedCompletion
  }
}

# Subscribe to contact engagement events
subscription ContactEngagement($contactId: UUID!) {
  contactEngagement(contactId: $contactId) {
    id
    type
    timestamp
    metadata
    score
  }
}
